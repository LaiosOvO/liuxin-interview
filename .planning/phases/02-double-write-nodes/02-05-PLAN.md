    ```python
    """StateGraph 构建 + 全局单例（Phase 2 完整版）。

    完整拓扑（PRD §4.1 DAG + 02-CONTEXT §4）：
      START → apply → manager_review ──reject→ END
                          ↓ advance
                    hr_initial ──return→ apply  ──reject→ END
                          ↓ advance（fan-out 5 并行节点）
              ┌──┬──┬──┬──┐
              ↓  ↓  ↓  ↓  ↓
            (5 parallel nodes — LangGraph 自动 fan-in)
              └──┴──┴──┴──┘
                          ↓ all done
                       hr_final ──reject→ END ──return→ device_return（v1 简化）
                          ↓ advance
              applicant_final_confirm ──return→ hr_final
                          ↓ advance
                       archive → END
    """
    from __future__ import annotations
    import logging
    from typing import Any

    from langgraph.graph import END, START, StateGraph

    from .checkpointer import make_checkpointer
    from .nodes import (
        APPLY_NODE_NAME, MANAGER_REVIEW_NODE_NAME,
        HR_INITIAL_NODE_NAME, HR_FINAL_NODE_NAME,
        APPLICANT_FINAL_CONFIRM_NODE_NAME, ARCHIVE_NODE_NAME,
        PARALLEL_NODES_META,
        apply_node, manager_review_node,
        hr_initial_node, hr_final_node,
        applicant_final_confirm_node, archive_node,
    )
    from .routes import (
        route_after_manager_review, route_after_hr_initial,
        route_after_hr_final, route_after_applicant,
        APPLY, HR_INITIAL, HR_FINAL, DEVICE_RETURN,
        APPLICANT_FINAL_CONFIRM, ARCHIVE, PARALLEL_NODES,
    )
    from .state import OffboardingState

    logger = logging.getLogger(__name__)

    _graph: Any | None = None


    def _route_after_hr_initial_to_parallel(state: OffboardingState):
        """hr_initial advance 时返回 list 触发 fan-out 到 5 并行节点。

        LangGraph 1.x 条件边路由函数可返回 list[str] 实现 dynamic fan-out。
        """
        from langgraph.types import Send

        action = state.get("current_action")
        if action == "reject":
            return END
        if action == "return":
            return APPLY
        # advance → fan-out 全部 5 并行节点
        # 每个 Send 都把当前 state 复制给目标节点
        return [Send(node, state) for node in PARALLEL_NODES]


    def _build_state_graph() -> StateGraph:
        """构建未 compile 的 StateGraph（Phase 2 完整 10 节点版）。"""
        b = StateGraph(OffboardingState)

        # 1. 注册全部 10 节点
        b.add_node(APPLY_NODE_NAME, apply_node)
        b.add_node(MANAGER_REVIEW_NODE_NAME, manager_review_node)
        b.add_node(HR_INITIAL_NODE_NAME, hr_initial_node)
        for name, _title, _desc, fn in PARALLEL_NODES_META:
            b.add_node(name, fn)
        b.add_node(HR_FINAL_NODE_NAME, hr_final_node)
        b.add_node(APPLICANT_FINAL_CONFIRM_NODE_NAME, applicant_final_confirm_node)
        b.add_node(ARCHIVE_NODE_NAME, archive_node)

        # 2. 入口边
        b.add_edge(START, APPLY_NODE_NAME)
        b.add_edge(APPLY_NODE_NAME, MANAGER_REVIEW_NODE_NAME)

        # 3. manager_review 三态条件边
        b.add_conditional_edges(
            MANAGER_REVIEW_NODE_NAME,
            route_after_manager_review,
            {END: END, HR_INITIAL: HR_INITIAL_NODE_NAME},
        )

        # 4. hr_initial 三态条件边 + advance 时 fan-out 5 并行节点
        b.add_conditional_edges(
            HR_INITIAL_NODE_NAME,
            _route_after_hr_initial_to_parallel,
            # mapping 列出所有可能目标（end / apply / 5 并行节点）
            [END, APPLY_NODE_NAME, *PARALLEL_NODES],
        )

        # 5. 5 并行节点 fan-in → hr_final（LangGraph 自动等待所有源节点 done）
        for name in PARALLEL_NODES:
            b.add_edge(name, HR_FINAL_NODE_NAME)

        # 6. hr_final 三态条件边
        b.add_conditional_edges(
            HR_FINAL_NODE_NAME,
            route_after_hr_final,
            {
                END: END,
                DEVICE_RETURN: PARALLEL_NODES[0],  # v1 简化：return 退回第一个并行节点
                APPLICANT_FINAL_CONFIRM: APPLICANT_FINAL_CONFIRM_NODE_NAME,
            },
        )

        # 7. applicant_final_confirm 两态条件边
        b.add_conditional_edges(
            APPLICANT_FINAL_CONFIRM_NODE_NAME,
            route_after_applicant,
            {
                HR_FINAL: HR_FINAL_NODE_NAME,
                ARCHIVE: ARCHIVE_NODE_NAME,
            },
        )

        # 8. archive → END
        b.add_edge(ARCHIVE_NODE_NAME, END)

        return b


    async def build_graph(use_memory_saver: bool = False) -> Any:
        global _graph
        if _graph is not None:
            return _graph
        builder = _build_state_graph()
        if use_memory_saver:
            from langgraph.checkpoint.memory import InMemorySaver
            saver: Any = InMemorySaver()
        else:
            saver = await make_checkpointer()
        _graph = builder.compile(checkpointer=saver)
        logger.info(
            "[graph] built and compiled with 10 nodes (saver=%s)",
            type(saver).__name__,
        )
        return _graph


    def get_graph() -> Any:
        if _graph is None:
            raise RuntimeError("graph not built yet — call build_graph() first")
        return _graph


    async def dispose_graph() -> None:
        global _graph
        _graph = None
    ```

    关键技术点：
    - **add_conditional_edges 的 mapping 可以是 list[str]**（LangGraph 1.x 接受 list 形式声明所有可能目标）
    - **使用 Send 实现 fan-out**：路由函数返回 `[Send(node, state) for node in PARALLEL_NODES]` 让 LangGraph 同时调度 5 个节点
    - **fan-in 自动**：因为 5 个节点都 `add_edge(name, HR_FINAL_NODE_NAME)`，LangGraph 默认 wait-all 语义

    3. 对每个 next 节点用 NodeRepository.upsert 写业务表（status=waiting_human）
    4. 这样前端再调 GET /api/flows/{id}/nodes 能立即看到新激活的节点 + node_state_id

    ```python
    async def submit_action(...):
        # ... Plan 01 双写规范（PENDING action_log + complete node + append node_results + commit）...
        if action == "reject":
            await self.flow_repo.mark_completed(flow_id, status=FlowStatus.REJECTED)
        await self.session.commit()
        # graph.ainvoke
        try:
            await self.graph.ainvoke(Command(resume=...), config=config)
        except Exception as e:
            # ... mark_failed + raise（Plan 01 已实现）...
            raise
        # 推进 graph 成功 → upsert next nodes（新激活的节点写业务表）
        await self._upsert_next_nodes_after_graph(flow_id, config)

        snapshot = await self.graph.aget_state(config)
        if not snapshot.next:
            # 已到 END
            async with self.session_factory() as final_session:
                final_flow_repo = FlowRepository(final_session)
                await final_flow_repo.mark_completed(flow_id, status=FlowStatus.COMPLETED)
                await final_session.commit()
        # mark_success action_log
        async with self.session_factory() as ok_session:
            ok_action_repo = ActionRepository(ok_session)
            await ok_action_repo.mark_success(action_log.id)
            await ok_session.commit()
        # 返回 flow 状态
        flow = await self.flow_repo.get(flow_id)
        next_nodes = [...]  # 从 snapshot.next 计算
        return {"flow_status": flow.status, "next_nodes": next_nodes, ...}

    async def _upsert_next_nodes_after_graph(self, flow_id, config):
        """graph.ainvoke 后，对每个 next 节点 upsert 业务表 waiting_human。"""
        snapshot = await self.graph.aget_state(config)
        next_names = list(snapshot.next or [])
        if not next_names:
            return
        # 用工厂新开一个 session（current self.session 已 commit 过）
        async with self.session_factory() as up_session:
            up_repo = NodeRepository(up_session)
            for name in next_names:
                # node_title 从 nodes/__init__.py 元数据查（avoid 写死）
                title = _resolve_node_title(name)
                await up_repo.upsert(
                    flow_id=flow_id, node_name=name,
                    node_title=title, status=NodeStatus.WAITING_HUMAN,
                )
            await up_session.commit()
    ```

    ```python
    import pytest
    from langgraph.graph import END, START

    from offboarding_flow.flow_engine.graph import _build_state_graph
    from offboarding_flow.flow_engine.nodes import (
        APPLY_NODE_NAME, MANAGER_REVIEW_NODE_NAME,

    }

    @pytest.fixture
    def builder():
        return _build_state_graph()

    def test_graph_has_10_nodes(builder):
        node_names = set(builder.nodes.keys())
        assert node_names == EXPECTED_10_NODES, f"got {sorted(node_names)}"

    def test_parallel_nodes_count(builder):
        parallel_names = [m[0] for m in PARALLEL_NODES_META]
        assert len(parallel_names) == 5
        for name in parallel_names:
            assert name in builder.nodes

    def test_graph_compiles_with_in_memory_saver():
        from langgraph.checkpoint.memory import InMemorySaver
        builder = _build_state_graph()

        # compiled 即可 — 进一步 invoke 测试在 test_e2e_full_flow.py
        assert graph is not None
    ```
