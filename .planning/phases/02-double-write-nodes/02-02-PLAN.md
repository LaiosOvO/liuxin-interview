                   "result_text": decision.get("result_text", ""),

    1. **routes.py**：
       ```python
       """路由函数集中模块（PRD §4.1 DAG + 02-CONTEXT §7）。

       条件边路由统一规则：
       - current_action == "reject" → 终止 (END)
       - current_action == "return" → 退回到上游指定节点
       - 其他（advance）→ 推进到下游节点
       """
       from __future__ import annotations
       import logging
       from langgraph.graph import END
       from offboarding_flow.flow_engine.state import OffboardingState

       logger = logging.getLogger(__name__)

       # 节点名常量（避免循环 import 写死字符串；与 nodes/__init__.py 保持一致）
       APPLY = "apply"
       MANAGER_REVIEW = "manager_review"
       HR_INITIAL = "hr_initial"
       DEVICE_RETURN = "device_return"
       ACCESS_REVOKE = "access_revoke"
       KNOWLEDGE_HANDOVER = "knowledge_handover"
       FINANCE_SETTLE = "finance_settle"
       LEGAL_SIGN = "legal_sign"
       HR_FINAL = "hr_final"
       APPLICANT_FINAL_CONFIRM = "applicant_final_confirm"
       ARCHIVE = "archive"

       PARALLEL_NODES = [DEVICE_RETURN, ACCESS_REVOKE, KNOWLEDGE_HANDOVER, FINANCE_SETTLE, LEGAL_SIGN]

       def route_after_manager_review(state: OffboardingState) -> str:
           action = state.get("current_action")
           logger.info("[route] after manager_review action=%s", action)
           if action == "reject":
               return END
           return HR_INITIAL  # advance / return → 下游 hr_initial（manager_review 是首人工节点无上游可退）

       def route_after_hr_initial(state: OffboardingState) -> str:
           action = state.get("current_action")
           logger.info("[route] after hr_initial action=%s", action)
           if action == "reject":
               return END
           if action == "return":
               return APPLY
           # advance → 推进到第一个并行节点；LangGraph 会因 add_edge 多目标自动 fan-out 全部
           return DEVICE_RETURN  # 实际并行靠 graph.py add_edge 多目标实现，本路由仅占位返回首节点

       def route_after_hr_final(state: OffboardingState) -> str:
           action = state.get("current_action")
           logger.info("[route] after hr_final action=%s", action)
           if action == "reject":
               return END
           if action == "return":
               return DEVICE_RETURN  # v1 简化：退回到第一个并行节点（重跑 5 个并行）
           return APPLICANT_FINAL_CONFIRM

       def route_after_applicant(state: OffboardingState) -> str:
           action = state.get("current_action")
           logger.info("[route] after applicant_final_confirm action=%s", action)
           if action == "return":
               return HR_FINAL
           return ARCHIVE  # advance（申请人无 reject）
       ```

       def _state(action):

           def test_advance_to_archive(self): assert route_after_applicant(_state("advance")) == ARCHIVE
           # 申请人无 reject — 即使误传 reject 也走 advance 路径
           def test_reject_falls_back_to_advance(self): assert route_after_applicant(_state("reject")) == ARCHIVE

  <action>
    复刻 backend/tests/test_flow_engine.py 的风格，针对 hr_initial / hr_final 写串行节点的 graph 集成单测（用 InMemorySaver）：
    ```python
    """test_nodes_serial.py — hr_initial 与 hr_final 节点 interrupt + Command(resume) 行为。

    本测试用 minimal StateGraph（只含两个被测节点），不接 graph.py 全拓扑。
    """
    from __future__ import annotations
    import uuid
    import pytest
    from langgraph.checkpoint.memory import InMemorySaver
    from langgraph.graph import END, START, StateGraph
    from langgraph.types import Command

    from offboarding_flow.flow_engine.nodes import (
        hr_initial_node, HR_INITIAL_NODE_NAME,
        hr_final_node, HR_FINAL_NODE_NAME,
    )
    from offboarding_flow.flow_engine.state import OffboardingState

    @pytest.fixture
    def hr_initial_graph():
        b = StateGraph(OffboardingState)
        b.add_node(HR_INITIAL_NODE_NAME, hr_initial_node)
        b.add_edge(START, HR_INITIAL_NODE_NAME)
        b.add_edge(HR_INITIAL_NODE_NAME, END)
        return b.compile(checkpointer=InMemorySaver())

    @pytest.fixture
    def hr_final_graph():
        b = StateGraph(OffboardingState)
        b.add_node(HR_FINAL_NODE_NAME, hr_final_node)
        b.add_edge(START, HR_FINAL_NODE_NAME)
        b.add_edge(HR_FINAL_NODE_NAME, END)
        return b.compile(checkpointer=InMemorySaver())

    async def _initial_state():
        return {
            "flow_id": str(uuid.uuid4()), "employee_id": "zhang.san",
            "current_action": None, "node_results": [], "context": {},
        }

    async def test_hr_initial_interrupts_for_decision(hr_initial_graph):
        state = await _initial_state()
        config = {"configurable": {"thread_id": state["flow_id"]}}
        result = await hr_initial_graph.ainvoke(state, config=config)
        # interrupt 后 graph 暂停 — result 应含 __interrupt__ 或 node_state 不含 hr_initial 的 done
        snapshot = await hr_initial_graph.aget_state(config=config)
        assert snapshot.next == (HR_INITIAL_NODE_NAME,)  # interrupt 后下一步仍是该节点（重跑等 resume）

    async def test_hr_initial_advance_resumes_to_end(hr_initial_graph):
        state = await _initial_state()
        config = {"configurable": {"thread_id": state["flow_id"]}}
        await hr_initial_graph.ainvoke(state, config=config)  # 走到 interrupt
        # resume advance
        final = await hr_initial_graph.ainvoke(
            Command(resume={"action": "advance", "result_text": "材料齐全", "actor": "hr.alice"}),
            config=config,
        )
        assert final["current_action"] == "advance"
        names = [r["node_name"] for r in final["node_results"]]
        assert HR_INITIAL_NODE_NAME in names

    async def test_hr_initial_return_resume(hr_initial_graph):
        state = await _initial_state()
        config = {"configurable": {"thread_id": state["flow_id"]}}
        await hr_initial_graph.ainvoke(state, config=config)
        final = await hr_initial_graph.ainvoke(
            Command(resume={"action": "return", "result_text": "材料不全", "actor": "hr.alice"}),
            config=config,
        )
        assert final["current_action"] == "return"

    async def test_hr_initial_reject_resume(hr_initial_graph):
        state = await _initial_state()
        config = {"configurable": {"thread_id": state["flow_id"]}}
        await hr_initial_graph.ainvoke(state, config=config)
        final = await hr_initial_graph.ainvoke(
            Command(resume={"action": "reject", "result_text": "材料造假", "actor": "hr.alice"}),
            config=config,
        )
        assert final["current_action"] == "reject"

    async def test_hr_final_advance(hr_final_graph):
        state = await _initial_state()
        config = {"configurable": {"thread_id": state["flow_id"]}}
        await hr_final_graph.ainvoke(state, config=config)
        final = await hr_final_graph.ainvoke(
            Command(resume={"action": "advance", "result_text": "通过", "actor": "hr.bob"}),
            config=config,
        )
        assert final["current_action"] == "advance"
        assert any(r["node_name"] == HR_FINAL_NODE_NAME for r in final["node_results"])

    async def test_hr_final_resume_with_string_decision_falls_back():
        """resume 传非 dict 时 — defensive fallback 到 advance。"""
        b = StateGraph(OffboardingState)
        b.add_node(HR_FINAL_NODE_NAME, hr_final_node)
        b.add_edge(START, HR_FINAL_NODE_NAME)
        b.add_edge(HR_FINAL_NODE_NAME, END)
        g = b.compile(checkpointer=InMemorySaver())
        state = await _initial_state()
        config = {"configurable": {"thread_id": state["flow_id"]}}
        await g.ainvoke(state, config=config)
        final = await g.ainvoke(Command(resume="raw string"), config=config)
        # defensive fallback：action=advance result_text="raw string"
        assert final["current_action"] == "advance"
    ```
