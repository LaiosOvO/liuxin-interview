async def submit_action(...):
    # 校验同 Phase 1
    # action_log 用 PENDING 状态创建
    action_log = await self.action_repo.create(..., status=ActionStatus.PENDING)
    await self.node_repo.complete(node_id, result_text, new_status)
    # 业务层 node_results 冗余
    await self.flow_repo.append_node_result(flow_id, {
        "node_name": node.node_name, "node_title": node.node_title,
        "result_text": result_text, "actor": actor,
        "completed_at": datetime.now(UTC).isoformat(),
    })
    # 若 reject → flow_instances.status=rejected
    if action == "reject":
        await self.flow_repo.mark_completed(flow_id, status=FlowStatus.REJECTED)
    await self.session.commit()
    # 推进 graph — 失败时新开 session mark action_log.failed + raise
    try:
        await self.graph.ainvoke(Command(resume={...}), config=config)
    except Exception as e:
        # 必须新开 session（原 session 已 commit）
        async with new_session_factory() as fail_session:
            fail_action_repo = ActionRepository(fail_session)
            await fail_action_repo.mark_failed(action_log.id, f"{type(e).__name__}: {e}")
            await fail_session.commit()
        logger.error("[node_service] graph.ainvoke failed for flow=%s action_log=%s: %s", flow_id, action_log.id, e)
        raise HTTPException(500, f"流程推进失败已记录 action_log={action_log.id}，可通过 recover_from_db.py 重试")
    # 成功 → mark_success
    async with new_session_factory() as ok_session:
        ok_action_repo = ActionRepository(ok_session)
        await ok_action_repo.mark_success(action_log.id)
        await ok_session.commit()
    # 返回 flow 状态

       ```python
       """scripts/recover_from_db.py — 从业务表重新驱动 LangGraph state（PITFALLS #2 修复路径）。

       用法：
         uv run python -m scripts.recover_from_db [--flow-id=UUID] [--dry-run] [--max-retries=3]
       """
       from __future__ import annotations
       import argparse, asyncio, logging, sys, uuid
       from langgraph.types import Command
       from offboarding_flow.flow_engine.graph import build_graph
       from offboarding_flow.state_store.session import new_session
       from offboarding_flow.state_store.repositories import ActionRepository
       from offboarding_flow.state_store.enums import ActionStatus

       logger = logging.getLogger("recover_from_db")

       async def recover_all(flow_id: uuid.UUID | None, dry_run: bool, max_retries: int) -> dict:
           stats = {"scanned": 0, "recovered": 0, "still_failed": 0, "skipped_max_retries": 0}
           graph = await build_graph()
           async with new_session() as session:
               action_repo = ActionRepository(session)
               failed = await action_repo.list_failed(limit=200, flow_id=flow_id)
           stats["scanned"] = len(failed)
           # In-memory retry count（action_logs 表无 retry_count 列；本 plan 不引迁移）
           # 多轮跑同一个 action 仍可——每次失败累加 in-memory counter
           retry_counts: dict[uuid.UUID, int] = {}
           for action in failed:
               if retry_counts.get(action.id, 0) >= max_retries:
                   stats["skipped_max_retries"] += 1
                   continue
               if dry_run:
                   logger.info("[dry-run] would recover action_log=%s flow=%s action=%s", action.id, action.flow_id, action.action)
                   continue
               try:
                   config = {"configurable": {"thread_id": str(action.flow_id)}}
                   await graph.ainvoke(
                       Command(resume={
                           "action": action.action,
                           "result_text": action.result_text or "",
                           "actor": action.actor,
                       }),
                       config=config,
                   )
                   async with new_session() as ok_session:
                       ok_repo = ActionRepository(ok_session)
                       await ok_repo.mark_success(action.id)
                       await ok_session.commit()
                   stats["recovered"] += 1
                   logger.info("[recover] action_log=%s flow=%s — success", action.id, action.flow_id)
               except Exception as e:
                   retry_counts[action.id] = retry_counts.get(action.id, 0) + 1
                   logger.exception("[recover] action_log=%s failed retry=%d: %s", action.id, retry_counts[action.id], e)
                   stats["still_failed"] += 1
           return stats

       def main(argv: list[str] | None = None) -> int:
           logging.basicConfig(level=logging.INFO, format="%(asctime)s %(name)s %(levelname)s %(message)s")
           parser = argparse.ArgumentParser()
           parser.add_argument("--flow-id", type=lambda s: uuid.UUID(s), default=None)
           parser.add_argument("--dry-run", action="store_true")
           parser.add_argument("--max-retries", type=int, default=3)
           args = parser.parse_args(argv)
           stats = asyncio.run(recover_all(args.flow_id, args.dry_run, args.max_retries))
           print(f"recover_from_db 完成: 扫描={stats['scanned']} 成功={stats['recovered']} 仍失败={stats['still_failed']} 跳过最大重试={stats['skipped_max_retries']}")
           return 0 if stats["still_failed"] == 0 else 1

       if __name__ == "__main__":
           sys.exit(main())
       ```
