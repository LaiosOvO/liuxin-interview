       class HulyListener:
           name = "huly"
           def __init__(self, settings):
               self._settings = settings
               self._dispatch: DispatchFn | None = None

           async def start(self) -> None:
               # sidecar 是独立容器，本 listener 不主动 connect；只承担 webhook → dispatch 转发
               logger.info("[huly_listener] 启动（webhook 模式，依赖 sidecar 反向推送）")

           async def stop(self) -> None:
               logger.info("[huly_listener] 停止")

           def register_command_listener(self, dispatch: DispatchFn) -> None:
               self._dispatch = dispatch

           async def handle_webhook(self, event: dict) -> None:
               """由 api/internal_huly.py 调用。event 含 sender_username / channel_id / channel_type / message"""
               if self._dispatch is None:
                   logger.warning("[huly_listener] dispatch 未注册，丢消息：%s", event.get('message','')[:80])
                   return
               # 构造 IMHelpers — 用 huly_im_provider 的 send_dm / post_to_channel
               from offboarding_flow.providers.factory import get_im_provider
               im = get_im_provider(self._settings)
               helpers = IMHelpers(
                   post_channel=im.post_to_channel,
                   send_dm=im.send_dm,
                   ensure_in_channel=im.ensure_user_in_channel,
               )
               # session_factory 从 state_store
               from offboarding_flow.state_store.session import new_session
               await self._dispatch(
                   sender_username=event['sender_username'],
                   user_id=event['sender_account'],
                   channel_id=event['channel_id'],
                   channel_type=event.get('channel_type', 'D'),
                   message=event['message'],
                   im_helpers=helpers,
                   settings=self._settings,
                   session_factory=new_session,
               )
       ```

       from fastapi import APIRouter, Header, HTTPException, Request
       from offboarding_flow.config import get_settings
       from typing import Annotated

       router = APIRouter(prefix="/api/internal/huly", tags=["internal-huly"])

       @router.post("/event")
       async def huly_event(
           request: Request,
           payload: dict,
           x_bridge_token: Annotated[str | None, Header(alias="X-Bridge-Token")] = None,
       ):
           settings = get_settings()
           if not settings.huly_bridge_token or x_bridge_token != settings.huly_bridge_token:
               raise HTTPException(status_code=401, detail="bridge_auth_failed")

           listener = request.app.state.huly_listener
           if listener is None:
               raise HTTPException(status_code=503, detail="huly_listener_not_initialized")
           await listener.handle_webhook(payload)
           return {"ok": True}
       ```
