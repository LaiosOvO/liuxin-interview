# backend/src/offboarding_flow/flow_engine/state.py
from operator import add
from typing import Annotated, TypedDict

class NodeResult(TypedDict):
    node_name: str
    node_title: str
    result_text: str
    actor: str
    completed_at: str  # ISO 8601

class OffboardingState(TypedDict):
    flow_id: str            # UUID4
    employee_id: str        # 离职员工 username
    current_action: str | None  # advance / return / reject — 最近一次决策
    node_results: Annotated[list[NodeResult], add]  # 累积所有节点结果，必须 reducer
    context: dict           # 流程上下文（员工信息等）

# backend/src/offboarding_flow/flow_engine/checkpointer.py 关键模式
from langgraph_checkpoint_postgres.aio import AsyncPostgresSaver
from psycopg_pool import AsyncConnectionPool
from psycopg.rows import dict_row

async def make_checkpointer(conninfo: str, schema_name: str = "langgraph") -> AsyncPostgresSaver:
    # PITFALLS #1: 必须 autocommit=True + row_factory=dict_row + prepare_threshold=0
    pool = AsyncConnectionPool(
        conninfo=conninfo,
        max_size=10,
        kwargs={
            "autocommit": True,
            "row_factory": dict_row,
            "prepare_threshold": 0,  # langgraph-checkpoint-postgres 强烈推荐
        },
    )
    # 等价于：saver = AsyncPostgresSaver(pool, schema_name=schema_name)
    return AsyncPostgresSaver(pool)  # schema 在 setup() 时由 saver 内部表名约定

       """LangGraph AsyncPostgresSaver 工厂 + CLI setup 入口。
       
       PITFALLS #1: psycopg 3 必须 autocommit=True + row_factory=dict_row + prepare_threshold=0
       否则 langgraph-checkpoint-postgres 在某些场景 deadlock。
       
       双 schema：LangGraph 数据放 `langgraph` schema（由 saver.setup() 自管），
       与 alembic 管理的 `app` schema 隔离（CONTEXT §2）。
       """
       from __future__ import annotations
       
       import argparse
       import asyncio
       import contextlib
       import logging
       from typing import AsyncIterator
       
       from langgraph.checkpoint.postgres.aio import AsyncPostgresSaver
       from psycopg.rows import dict_row
       from psycopg_pool import AsyncConnectionPool
       
       from offboarding_flow.config import get_settings
       
       logger = logging.getLogger(__name__)
       
       _saver: AsyncPostgresSaver | None = None
       _pool: AsyncConnectionPool | None = None
       
       
       async def make_checkpointer(conninfo: str | None = None) -> AsyncPostgresSaver:
           """创建（或返回已有）AsyncPostgresSaver 单例。"""
           global _saver, _pool
           if _saver is not None:
               return _saver
           
           if conninfo is None:
               settings = get_settings()
               conninfo = settings.langgraph_pg_conninfo
           
           _pool = AsyncConnectionPool(
               conninfo=conninfo,
               max_size=10,
               min_size=2,
               kwargs={
                   "autocommit": True,
                   "row_factory": dict_row,
                   "prepare_threshold": 0,
               },
               open=False,
           )
           await _pool.open()
           _saver = AsyncPostgresSaver(_pool)
           return _saver
       
       
       async def setup_checkpointer_schema() -> None:
           """显式跑 saver.setup() — 创建 langgraph schema 的所有表。
           
           entrypoint.sh 在启动 uvicorn 前调用本方法。
           幂等 — 重复调用不会报错（langgraph 内部用 CREATE TABLE IF NOT EXISTS）。
           """
           saver = await make_checkpointer()
           logger.info("[checkpointer] setup() — creating LangGraph schema tables")
           await saver.setup()
           logger.info("[checkpointer] setup() complete")
       
       
       async def dispose_checkpointer() -> None:
           """关闭 pool — 用于 FastAPI shutdown。"""
           global _saver, _pool
           if _pool is not None:
               await _pool.close()
               _pool = None
               _saver = None
       
       
       def _run_setup_cli() -> None:
           """CLI 入口：`python -m offboarding_flow.flow_engine.checkpointer --setup`."""
           parser = argparse.ArgumentParser(description="LangGraph checkpointer setup")
           parser.add_argument("--setup", action="store_true", help="跑 saver.setup() 创建 langgraph schema 表")
           args = parser.parse_args()
           
           if args.setup:
               logging.basicConfig(level=logging.INFO, format="%(asctime)s [%(name)s] %(levelname)s %(message)s")
               asyncio.run(_setup_then_dispose())
           else:
               parser.print_help()
       
       
       async def _setup_then_dispose() -> None:
           try:
               await setup_checkpointer_schema()
           finally:
               await dispose_checkpointer()
       
       
       if __name__ == "__main__":
           _run_setup_cli()
       ```

       """StateGraph 构建 + 全局单例。
       
       Phase 1 最小图：START → apply → manager_review → END
       Phase 2 才扩展到 10 节点 + 并行 fan-out/fan-in。
       """
       from __future__ import annotations
       
       import logging
       from typing import TYPE_CHECKING
       
       from langgraph.checkpoint.memory import InMemorySaver
       from langgraph.graph import END, START, StateGraph
       
       from .checkpointer import make_checkpointer
       from .nodes import (
           APPLY_NODE_NAME,
           MANAGER_REVIEW_NODE_NAME,
           apply_node,
           manager_review_node,
       )
       from .state import OffboardingState
       
       if TYPE_CHECKING:
           from langgraph.graph.state import CompiledStateGraph
       
       logger = logging.getLogger(__name__)
       
       _graph: "CompiledStateGraph | None" = None
       
       
       def _route_after_manager(state: OffboardingState) -> str:
           """manager_review 后的路由：reject 终止；advance/return 在 Phase 1 也终止（下一节点 Phase 2 加）。"""
           action = state.get("current_action")
           if action == "reject":
               logger.info("[route] manager_review rejected → END")
               return END
           # Phase 1: advance/return 都终止（hr_initial Phase 2 才加）
           logger.info("[route] manager_review %s → END (Phase 1 stops here)", action)
           return END
       
       
       def _build_state_graph() -> StateGraph:
           """构建未 compile 的 StateGraph。"""
           builder = StateGraph(OffboardingState)
           builder.add_node(APPLY_NODE_NAME, apply_node)
           builder.add_node(MANAGER_REVIEW_NODE_NAME, manager_review_node)
           
           builder.add_edge(START, APPLY_NODE_NAME)
           builder.add_edge(APPLY_NODE_NAME, MANAGER_REVIEW_NODE_NAME)
           builder.add_conditional_edges(MANAGER_REVIEW_NODE_NAME, _route_after_manager, {END: END})
           
           return builder
       
       
       async def build_graph(use_memory_saver: bool = False):
           """构建并 compile graph。production 用 AsyncPostgresSaver，测试用 InMemorySaver。"""
           global _graph
           if _graph is not None:
               return _graph
           
           builder = _build_state_graph()
           
           if use_memory_saver:
               saver = InMemorySaver()
           else:
               saver = await make_checkpointer()
           
           _graph = builder.compile(checkpointer=saver)
           logger.info("[graph] built and compiled (saver=%s)", type(saver).__name__)
           return _graph
       
       
       def get_graph():
           """返回已 build 的 graph；若未 build 抛错。"""
           if _graph is None:
               raise RuntimeError("graph not built yet — call build_graph() first")
           return _graph
       
       
       async def dispose_graph() -> None:
           """清理 graph 单例（用于 shutdown / 测试 teardown）。"""
           global _graph
           _graph = None
       ```
