     ```python
     """E2E: PRD §6.2 + PITFALLS #6 — 双击 race condition。

     验收 ROADMAP §Phase 3 Success Criteria #1:
     asyncio.gather(exchange(token), exchange(token)) 仅 1 个 200，另一个 401

     本测试用 ASGITransport 不需要 real container，故无 e2e marker（pytest 默认运行）。
     """
     import asyncio
     import time
     import uuid
     import httpx
     import pytest
     import pytest_asyncio
     from asgi_lifespan import LifespanManager
     from offboarding_flow.auth import encode, JWTPayload

     pytestmark = pytest.mark.asyncio

     @pytest_asyncio.fixture
     async def app_client():
         from offboarding_flow.main import create_app
         app = create_app()
         async with LifespanManager(app):
             async with httpx.AsyncClient(transport=httpx.ASGITransport(app=app), base_url="http://test") as c:
                 yield c

     async def test_20_concurrent_exchanges_only_one_wins(app_client, sample_node_and_user, redis_client):
         """20 个并发 exchange 同 token，仅 1 个 200 / 19 个 401。"""
         payload = JWTPayload(
             sub="li.si", email="li.si@demo.local", role="manager",
             flow_id=sample_node_and_user["flow"].id,
             node_id=sample_node_and_user["node"].id,
             node_name="manager_review",
             allowed_actions=["advance", "return", "reject"],
             iat=int(time.time()),
             exp=int(time.time()) + 3600,
             jti=uuid.uuid4().hex,
         )
         token = encode(payload)

         # 20 并发 POST
         async def call():
             return await app_client.post("/api/auth/exchange", json={"token": token})

         responses = await asyncio.gather(*[call() for _ in range(20)])
         status_200 = sum(1 for r in responses if r.status_code == 200)
         status_401 = sum(1 for r in responses if r.status_code == 401)

         assert status_200 == 1, f"期望仅 1 个 200，实际 {status_200}"
         assert status_401 == 19, f"期望 19 个 401，实际 {status_401}"
     ```

  3. `backend/tests/e2e/test_cross_flow_rejection.py`：验收 SC #5 — 跨 flow_id token 立即拒绝
     ```python
     """E2E: 跨 flow_id 复用 token 立即拒绝。

     验收 ROADMAP §Phase 3 Success Criteria #5
     场景：flow_a 的 token 试图用在 flow_b 的 node 上 → 401（cross_flow reason）
     """
     import time
     import uuid
     import httpx
     import pytest
     import pytest_asyncio
     from asgi_lifespan import LifespanManager
     from offboarding_flow.auth import encode, JWTPayload

     pytestmark = pytest.mark.asyncio

     async def test_token_for_flow_a_rejected_on_flow_b(app_client, two_flows_with_nodes, redis_client):
         """token 的 flow_id=A 但 URL 携带 node 属于 flow_id=B → exchange 401。

         实现：token 用 flow_a 签名（payload.flow_id=A），但 node_id=B_node（属于 flow_b）
         → exchange_token 第 3 步 str(node.flow_id) != str(payload.flow_id) → AuthError(reason='cross_flow')
         """
         flow_a = two_flows_with_nodes["flow_a"]
         flow_b = two_flows_with_nodes["flow_b"]
         node_b = two_flows_with_nodes["node_b"]

         # token: flow_id=A，但 node_id=B_node（来自 flow_b）
         payload = JWTPayload(
             sub="li.si", email="li.si@demo.local", role="manager",
             flow_id=flow_a.id,     # 错的 flow_id
             node_id=node_b.id,     # node 属于 flow_b
             node_name="manager_review",
             allowed_actions=["advance"],
             iat=int(time.time()),
             exp=int(time.time()) + 3600,
             jti=uuid.uuid4().hex,
         )
         token = encode(payload)

         resp = await app_client.post("/api/auth/exchange", json={"token": token})
         assert resp.status_code == 401
     ```
