  8. `backend/src/offboarding_flow/auth/jwt_service.py`：encode/decode 实现
     ```python
     """JWT 签发与解码 — HS256 + JWT_SECRET（仅 env 注入）。

     约定（CONTEXT §D4）：
     - encode 自动写 iat/exp/jti（如果 payload 未填）
     - decode 严格校验签名 + exp + 必需字段；不允许 leeway
     - 对外异常用 auth.errors 体系（不暴露 pyjwt 内部异常类型）
     """
     from __future__ import annotations
     import time
     import uuid
     from typing import Any
     import jwt as pyjwt  # pyjwt 包名 jwt，alias 防与本模块冲突
     from pydantic import ValidationError
     from offboarding_flow.config import get_settings
     from .errors import ExpiredTokenError, InvalidTokenError
     from .schemas import JWTPayload, SessionPayload

     _ALG = "HS256"

     def encode(payload: JWTPayload) -> str:
         """签发 token。若 payload.iat/exp/jti 未设置，由本方法兜底（测试时通常显式传）。"""
         settings = get_settings()
         data = payload.model_dump(mode="json")
         return pyjwt.encode(data, settings.jwt_secret, algorithm=_ALG)

     def decode(token: str) -> JWTPayload:
         """解码 + 校验 + Pydantic 校验。失败抛 AuthError 子类。"""
         settings = get_settings()
         try:
             raw: dict[str, Any] = pyjwt.decode(
                 token,
                 settings.jwt_secret,
                 algorithms=[_ALG],
                 options={"require": ["exp", "iat", "jti", "sub"]},
                 leeway=0,
             )
         except pyjwt.ExpiredSignatureError as e:
             raise ExpiredTokenError("token 已过期") from e
         except pyjwt.InvalidTokenError as e:
             raise InvalidTokenError("token 无效") from e
         try:
             return JWTPayload.model_validate(raw)
         except ValidationError as e:
             raise InvalidTokenError(f"token payload schema 不匹配") from e

     def encode_session(token_payload: JWTPayload) -> str:
         """从 token payload 派生 session cookie 值。"""
         settings = get_settings()
         exp = int(time.time()) + settings.session_expiry_hours * 3600
         session = SessionPayload(
             sub=token_payload.sub,
             role=token_payload.role,
             flow_id=token_payload.flow_id,
             node_id=token_payload.node_id,
             exp=exp,
         )
         return pyjwt.encode(session.model_dump(mode="json"), settings.jwt_secret, algorithm=_ALG)

     def decode_session(cookie_value: str) -> SessionPayload:
         """解码 session cookie。失败抛 AuthError 子类。"""
         settings = get_settings()
         try:
             raw = pyjwt.decode(
                 cookie_value,
                 settings.jwt_secret,
                 algorithms=[_ALG],
                 options={"require": ["exp", "sub"]},
                 leeway=0,
             )
         except pyjwt.ExpiredSignatureError as e:
             raise ExpiredTokenError("session 已过期") from e
         except pyjwt.InvalidTokenError as e:
             raise InvalidTokenError("session 无效") from e
         try:
             return SessionPayload.model_validate(raw)
         except ValidationError as e:
             raise InvalidTokenError("session payload schema 不匹配") from e
     ```

  9. `backend/src/offboarding_flow/auth/deep_link.py`：build_deep_link
     ```python
     """深链 URL 构造 — R2 方案 A query string 格式（PRD §6.2 + SUMMARY R2）。

     格式：{base_url}/flow/handle?flow_id=xxx&node_id=yyy&token=zzz
     不用 path param（Next.js 15 issue #79380）；不用 fragment（不发服务器）。
     """
     from __future__ import annotations
     from urllib.parse import urlencode
     from .schemas import JWTPayload

     _HANDLE_PATH = "/flow/handle"

     def build_deep_link(token: str, payload: JWTPayload, *, base_url: str) -> str:
         """构造一键登录深链 URL。

         Args:
             token: JWT 字符串（已编码）
             payload: JWT payload（用于提取 flow_id / node_id 拼到 query）
             base_url: DEEPLINK_BASE_URL（无尾斜杠）
         Returns:
             完整 URL 例：http://192.168.2.44:3000/flow/handle?flow_id=...&node_id=...&token=...
         """
         base = base_url.rstrip("/")
         query = urlencode({
             "flow_id": str(payload.flow_id),
             "node_id": str(payload.node_id),
             "token": token,
         })
         return f"{base}{_HANDLE_PATH}?{query}"
     ```
